buat table sessions;
id biginteger auto_increment primary key;
token varchar(255) not null; (isinya uuid untuk token user login)
user_id bigint not null; (FK dari tabel users)
created_at timestamp default current_timestamp;
updated_at timestamp default current_timestamp on update current_timestamp;

buatkan API untuk login user dengan expired 1 hari

API ini akan menerima request 
POST /login

dengan body : 
{
    "email": "[EMAIL_ADDRESS]",
    "password": "[PASSWORD]"
}

response json status 200 OK
{
    "success": true,
    "message": "User berhasil login",
    "code": 2000,
    "data": {
        "token": "jwt"
        "issuedAt": "timestamp",
        "expiresAt": "timestamp"
    }
}

response error
{
    "success": false,
    "message": "Email atau password salah",
    "code": 4000,
    "error": []
}

response error validasi input
{
    "success": false,
    "message": "Invalid input",
    "code": 4001,
    "error": [
        {
            "field": "email",
            "message": "Email tidak boleh kosong"
        },
        {
            "field": "password",
            "message": "Password tidak boleh kosong"
        }
    ]
}
